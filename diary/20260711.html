            y[x[r]] = -1;

E - Range Flip
(46'15)


int main(void){
    int n, k; cin >> n >> k;

    vector> a(n);
    int i;
    rep(i,n) cin >> a[i].first >> a[i].second;

    vector> dp(n+1, vector(k+1,0));

    ll x=0, y=0;
    int j; i=0;
    rep(j,n){
        x += a[j].first;
        y = max(y+a[j].second, x);
        dp[j+1][i] = dp[j][i];
        dp[j+1][i] = max(dp[j+1][i], y);
    }
    
    rep(i,k){
        if(i==0) continue;
        x=0;y=0;
        ll z= 0;    
        rep(j,n){
            x = dp[j+1][i-1];
            y = max(y+a[j].first, x);
            z = max(z+a[j].second, y);

            dp[j+1][i] = dp[j][i];
            dp[j+1][i] = max(dp[j+1][i], z);
        }
    }
    ll ans = 0;

    rep(i,n){
        ans = max(ans + a[i].first, dp[i+1][k-1]);
    }
    cout << ans;

}

区間が被るような二つの範囲のflipは、被らないような二つに組み替えられる（公式解説参照）ので、 被らないk個までの区間としてOK
dp[j][i]で（これまでのflip回数i回、現在見ているカード1~jとして）1~jでi回までflipしたときの最大スコア（ただし、カードjは裏返しになるようにflip） とすれば、遷移が上のコードみたいに書けるので、あとは、どこでflipを終えているべきか探す

F - Many Mod Calculation
(92'02)


int main(void){
    int t; cin >> t;
    for(;t--;){
        ll n, x; 
        cin >> n >> x;
        vector a(n);
        int i;
        rep(i,n) cin >> a[i];
        vector b; b.reserve(n);
        b.push_back(x+1);

        rep(i,n){
            if(b.empty() || (*b.rbegin()) > a[i]){
                b.push_back(a[i]);
            } 
        }
        n = b.size();
        
        vector ans(n+1,0);
        ans[n] = 1;
        rep(i,n-1){
            ll x = b[n-i-1], y = b[n-i-2]; //y>x;
            if(i==0){
                ans[n-i-1] += (y-1)/x +1;
            }else{
                ans[n-i-1] += ((y-1)/x ) * (ans[n-i]);
                ll rem = y%x;
                while(rem>0){
                    int l = n-i-1, r = n; // l ng / r ok
                    while(r-l > 1){
                        int m = (r+l)/2;
                        if(rem >= b[m]) r = m;
                        else l = m;
                    }
                    if(r == n) {ans[n-i-1]++; break;}
                    ans[n-i-1] += ((rem)/b[r]) * (ans[r+1]);
                    rem %= b[r];
                }
            }

        }
        cout << ans[1]-1 << "\n";


    }
}

Aの先頭にX+1をくっつけると、あとで楽です
まずは、Aを降順にスターリンソート（粛清ソート）します 後に出てくる大きい数字は計算結果に何ら影響を及ぼさないので
あと、コードではこれは配列bに該当します
そしたら、後ろから順に、a[i], a[i+1]についての部分問題

\(0\leq x < a[i]\)を満たすxのうち、\( ((x \mod a[i+1]) \mod a[i+2])... \mod a[n-1] = 0\)
を満たすxの数を求めてください
を解いていくと、答え+1 (0そのものは答えとしてカウントされないので)が得られて、ACできます
これの解き方は、かなり感覚でやってて言語化に1年かかりそうなので、書かない

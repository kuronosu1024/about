A - FourColoring

int main(void){
    int n; cin >> n;

    int i, j;
    array<int,4> p = {2,4,1,3};
    rep(i,n){
        rep(j,n){
            int x; cin >> x;
            cout << p[x-1] << " ";
        }
        cout << "\n";
    }
}

最初は与えられたものに対して構造を考えながらじゃないと解けないかと思ってたけど、順位表見たらそんなことなさそうだったので簡単な方法を探しに行った

B - Independent Nim
(53'42 3x)


int main(void){
    int t; cin >> t;
    for(;t--;){
        int n; cin >> n;
        bool ae = true;
        int p = 0;
        int i;
        rep(i,n){
            int a; cin >> a;
            if(a==1){
                p++;
            }else{
                if(p!=2 && p!=0) ae = false;
                p=0;
            }
        }
        if(p!=2 && p!=0) ae = false;

        if(ae) cout << "Bob\n";
        else cout << "Alice\n";
    }
}

まず、1の連結成分ごとに別の山っぽいなと思って、それぞれの連結成分の長さに対してどういう勝ち負けへの寄与があるかを考えた
そしたらいつの間にか脳内では、各連結成分から一個だけ取れますと勘違いしだして、長さが偶数だけの盤面を送り続ければ勝ちと誤認したコードを投げて+1ペナ
勘違いに気付いて、長さが2のものだけならOKと正解にたどり着いたが、長さ0があった時を弾いていて+2ペナ

D - Gap Swap (easy)
(75'26)


int main(void){
    int n; cin >> n;
    vector<int> p(n);
    int i;
    rep(i,n) cin >> p[i];

    ll ans = 0;
    rep(i,n){
        ans += abs(p[i]-1 -i);
    }
    ans/=2;
    cout << ans << "\n";
}

割と早めに下限\(\sum_{i=0}^n |P_i - i|\)に気付いたものの、確信が持てずわやわやしてたら、
「揃ってなければ操作し得な境目が存在する （P_i - i は左端は正、右端は負で、0を無視したらどっかに正→負の境目があって、そこ）」
と気づけたため何とかAC 下限でいけるかな？と思ったときに一回投げてみるべきなのかな

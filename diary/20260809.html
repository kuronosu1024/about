A - Meeting Division
(7'10)


int main(void){
    int n; cin >> n;
    vector> u(n);
    int i;
    rep(i,n) cin >> u[i].first >> u[i].second;
    sort(all(u));

    mint ans = 1;

    set d;
    int t=0, c=0;
    rep(i,n){
        t = u[i].first;
        while(!d.empty() && *d.begin() <= t){
            d.erase(d.begin());
            c--;
        }
        if(c==0) ans*=2;
        else if(c>1) ans *= 0;
        c++;
        d.insert(u[i].second); 
    }
    cout << ans << "\n";
}

各会議を \(N\) 頂点のグラフ \(G\) の頂点に対応させて、開催時間を共有する2つの会議どうしを辺でつなぐと、こういう問題になる↓

  • \(G\) の各頂点を2色 (高橋君と青木君) で塗り分ける方法は何通りあるか。
これは、\(G\) が2部グラフなら（\(S = Gの連結成分数\) とすると）答えは \(2^S\) で、 そうでなければ \(0\) です(証明は2部グラフが分かれば自明だと思います)
2部グラフでないことは、今回の問題設定では「同時に3つ以上の会議が開催されている」ことと同値なので、これで解けます

B - Bin-ary Packing
(28'19)


int main(void){
    int T; cin >> T;
    for(;T--;){
        int n,m ; cin >> n >> m;
        ll mx=0, mc=0, sx=0, sc=n, ux = -1;
        vector a(m); int i;
        rep(i,m) cin >> a[m-i-1];
        ll h = 1ll<<(m);

        rep(i,m){
            h>>=1;
            if(ux != -1){
                if(mx - ux > h*a[i]) {
                    ux += h*a[i];
                    continue;
                }else{
                    a[i] -= (mx-ux)/h;
                    ux = -1;
                    mc++;
                }
            }
            if(mc != 0){
                ll p = (mx-sx)/h;
                if(p*sc > a[i]){
                    sc -= a[i]/p;
                    mc += a[i]/p;
                    if(a[i]%p > 0){
                        sc--;
                        ux = sx + h*(a[i]%p);
                    }
                    a[i] = 0;
                    continue;
                }else{
                    a[i]-=p*sc;
                    mc=0;
                    sc=n;
                    sx=mx;
                }
            }
            if(mc == 0){
                sx += h*(a[i]/n);
                mx = sx + h*(a[i]%n!=0);
                mc = a[i]%n;
                sc = n-mc;
            }
        }
        cout << mx << "\n";
    }
}

想定解は解の2分探索らしいが、maspyさんのユーザ解説と(ほぼ)同じ重いほうから振り分けるやつで解いた
具体的には、全ての荷物について重い順に

  • ①袋の重さがすべて同じとき
    • 分けなおすことなくそのままどれかに入れる
  • ②そうでないとき
    • 一番重くない袋のうち最も重いものに入れる
正当性については、荷物の総数を \(S\) 、重い順に前からk個の荷物に対する最適解を \(X_k\) としたとき、

①について、その袋の重さ（＝最適解）を \( X_k \) とし、それに重さ \(2^a\) の荷物を一つ追加するとき
  • 既に分けられた荷物のそれぞれの重さは全て \(2^a\) の倍数である。今追加する荷物を含めた全ての荷物を重さ \(2^a\) に分解した問題の解は \(X_k + 2^a\) で、分解する前の問題の最適解はこれ以上である
  • この最適解の下限 \(X_k + 2^a\) は、前述した操作で達成可能である
よって \(X_{k+1}=X_k + 2^a \) であり、操作は正当

②について、それまでの分け方が最適であると仮定し、それを \(X_k\) としてそこに重さ \(2^a\) の荷物を一つ追加するとき
  • 一番重くない袋のうち最も重いものの重さを \(W\) としたとき、既に分けられた荷物のそれぞれの重さは全て \(2^a\) の倍数であるので、\(X_k - W\) もまた \(2^a\) の倍数である。よって \(X_k - W \ge 2^a\) であり、操作によって袋の重さが \(X_k\) を超えることはない
  • また、\(X_i \le X_j\) \((i\lt j)\) （なぜならば、最適解 \(X_j\) の構成方法からいくつか取り除くことで、明らかに \(X_j\) 以下の解 \(X_i\) が得られるため）
よって、\(X_{k+1}=X_k\) であり、操作は正当

\(k=0\) では \(X_k = 0\) なのは疑う余地もないので、そこから上記の貪欲をすることで最適解を更新していける
また、これによって生まれる袋の重さの種類は最大3種類で、そのうち間の重さのものは1個しか生じ得ないのでそれぞれの個数と重さを管理することで高速化が可能
→これによって\(O(M)\)で解けます

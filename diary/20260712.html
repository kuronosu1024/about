        ll n; cin >> n;

            ll m = (r+l)/2;

            else r=m;

            ans += 2*4;

            ans += 4*(x-2)*(x-2);
        }

        ll rem = n - x*x;
        if(rem == 0);
        else if(rem <= x){
            ans += rem;

            if(rem == 1) ans += 1;
            else{
                ans += 2*2;
                ans += 3*(rem-2);
            }
        }else{
            ans += x;

            if(x==1){
                ans += 1;
            }else{
                ans += 2*2;
                ans += 3*(x-2);
            }
            rem -= x;

            ans += rem;

            if(rem == 1) ans += 1;
            else{
                ans += 2*2;
                ans += 3*(rem-2);
            }
        }

        cout << ans/2 << "\n";
    }

この文章書いてる今になって証明思いついたわ
ある\(n\)枚のタイル配置において、条件を満たすタイルのペアの個数を\(X\)、タイル配置の周長（内側の空白部分も含む）を\(Y\)とすると、 \(2X = 4n - Y\)が言えて、このことより\(Y\)を最小化するべき
内側の空白部分は明らかに無駄なので、そうしないことにすると、タイル配置が収まる最小の長方形が\(a\times b\)のとき、\(Y=2(a+b)\)である
あとは\(ab\ge n\)なる\(a,b\)についての\(a+b\)の最小化問題で、これは、知っとるやろ（説明放棄）

実数の時は相加相乗でやるやつ　今回は自然数だから、ちょっとめんどい

void slv(vector<vector<int>> &nb, vector<int> &val, int x){
    for(auto nx : nb[x]){
        if(val[nx] == -2){
            val[nx] = val[x]+1;
            slv(nb,val,nx);

}

ll pw(ll a, ll b){
    ll ret = 1;
    while(b>0){
        if(b&1) ret *= a;
        b>>=1;
        a*=a;
    }
    return ret;
}
int main(void){
    int t; cin >> t;
    for(;t--;){
        ll n,k;cin >>  n >> k;
        if(n<60 && (1ll<<n) < k) {cout << "-1\n"; continue;}
        vector<int> c(7,0); //c[i] -> 書き込みにコストiが必要な値の数
        ll K = k;
        int i=0;
        int j;
        while(K>0){
            K/=10;
            i++;
        }
        c[i] = k-pw(10,i-1);
        rep(j,i){
            if(j==0) continue;
            c[j] = pw(10,j)-pw(10,j-1);
        }

        ll ans = 0;
        j = 1;
        ll cap = n, Cap = n;
        while(true){
            //j回書く数
            if(cap < c[i]){
                ans += i*cap*j;
                c[i]-=cap;

                
                Cap *= (n-j);
                j++;
                Cap /= j;
                cap = Cap;
            }else{
                while(cap >= c[i]){
                    ans += i*c[i]*j;
                    cap -= c[i];
                    c[i] = 0;
                    i--;
                    if(i==0) break;
                }
                if(i==0) break;
            }
        }
        cout << ans << "\n";
    }
}

まず、1～\(K\)のうちどれか一つは書かなくてもよいので、\(K\)は書かないことにする
書き入れる数字にそれぞれ\(n\)桁の2進数を対応させて、 それの\(k\)桁目が1の時に丁度カード\(k\)にその数字を書き入れるとすると、 どっちかのみが存在するカードが必ず1つ以上存在して、逆に、2進数の対応が間に合わないなら不可能
あとは、コストが重い数字の書く量が少なくなるように割り当てればいい

部分文字列"ABC","ABB","AC"を見つけたらそれぞれ消せるだけ消していい
①"ABC"については、BCやCを残して消したことで新しく消せるようになる部分は存在しない、AABC, ABABCのように、順序が変わるだけだから
②"ABB"については、後ろのBが邪魔で新しくCが消せる位置に来ることはなく、また、Aだけで消す意味も①と同様にない
③"AC"についても同様

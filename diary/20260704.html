C : Reverse Permutation
(13'25)


int main(void){
    int n; cin >> n;
    string s; cin >> s;

    vector<ll> ans(n);
    int i;
    int l=0, r = n-1; bool b = false;

    rep(i,n){
        int x = n-i-1;
        if(s[x] == 'o'){
            b=!b;
            if(b){
                ans[l] = x; l++;
            }else{
                ans[r] = x; r--;
            }
        }else{
            if(b){
                ans[l] = x; l++;
            }else{
                ans[r] = x; r--;
            }
        }
    }
    rep(i,n) cout << ans[i]+1 << " ";
}

末尾の数字\(N\)の位置は\(s[N-1]=\)'o'なら左端、でなければ右端
反転したかのフラグを持ちながら、末尾から順に先頭まで位置を確定させていく

D : X to Y
(20'39)


int main(void){
    int t ; cin >> t;
    for(;t--;){
        ll x, y, k; cin >> x >> y >> k;
        ll ans = 0;
        while(x!=y){
            if(x<y) swap(x,y);
            x/=k;
            ans++;
        }
        cout << ans << "\n";
    }
}

\(x\)について逆操作ができるので、ゴールのほうも置き換えていってよいとできる
あと、操作を言い換えると
\(x=yk+\alpha\) \((0\leq \alpha< k )\)
\(y=xk+\alpha\) \((0\leq \alpha< k )\)
を満たす\(y\)に変換ができます、ということなので、\(x\)と\(y\)をでかくして合流させようとすると\(k\)で割った商が一生一致しないと分かるから、\(x\)も\(y\)も小さくしましょう、一致したら終わり

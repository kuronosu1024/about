Eがぱっと見無理そうだったので、先にFを解いた
今回の \(10^{100}\) は無限じゃなくてどちらかといえば \(2^{\infty}\)

swap(a,b) を \(n\) 頂点の無向グラフ \(G\) の頂点a,b間の辺と捉えると見通しがよい
一旦 \(10^{100}\) までの自由な回数swapしてよいことにすると、\(G\) で連結な頂点を自由に並び替えられる（バブルソートの要領で）
→この答えは各連結成分で重複順列の個数を数えた総積になる

元の丁度 \(10^{100}\) 回に戻すと、全ての連結成分において重複がなかった時に2倍ずれる

総操作回数は実質 \(2^{\infty}\) で、この時操作後の文字列の転倒数の偶奇は操作前と等しくなるので、そうならないような文字列を含めてはならない
→転倒数が偶数の順列と奇数の順列は同じ数あります(1,2,3,4 → 2,3,1,4 みたいな感じで全単射の写像が作れるので)

class AreaRect {
	map<ll,ll> M;
public:
	ll sum;
	AreaRect() {
		M[0]=1LL<<60;
		M[1LL<<60]=0;
		sum=0;
	}
	void add(ll x,ll y) {
		auto k=M.lower_bound(x);
		if(k->second>=y) return;
		while(prev(M.lower_bound(x))->second<=y) {
			auto p=*prev(M.lower_bound(x));
			M.erase(p.first);
			sum-=(p.first-prev(M.lower_bound(p.first))->first)*(p.second-M.lower_bound(p.first)->second);
		}
		sum += (x-prev(M.lower_bound(x))->first)*(y-M.lower_bound(x)->second);
		M[x]=y;
	}
};

C - Inc, Dec, Xor
(70'10)


int main(void){
    int n, q; cin >> n >> q;
    vector<int> a(n,0);
    set<int> p;
    ll ans = 0;
    int i;
    for(;q--;){
        int r,x; cin >> r;
        if(r==1){
            cin >> x; x--;
            ans ^= a[x];
            if(a[x] == 0){
                p.insert(x);
            }
            a[x]++;
            ans ^=a[x];
        }else{
            queue<int> te;
            for(auto u : p){
                ans ^= a[u];
                a[u]--;
                ans ^= a[u];
                if(a[u]==0) te.push(u);
            }
            while(!te.empty()){
                p.erase(te.front()); te.pop();
            }
        }
        cout << ans << "\n";
    }
}

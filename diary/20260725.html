    bool st = false;
    bool ed = false;

    do{
        if(a == q) ed = true;

D - Pre-Palindrome
(24'15)


int main(void){
    string s; cin >> s;

    int n = s.size();

    int i,j;
    ll ans = 0;
    rep(i,n){
        int c = 0;
        ans++;
        rep(j,min(i,n-i-1)){
            if(s[i-j-1] != s[i+j+1]) c++;
            if(c>1) break;
            ans++;
        }
    }
    rep(i,n-1){
        int c=0;
        rep(j,min(i+1,n-i-1)){
            if(s[i-j] != s[i+j+1]) c++;
            if(c>1) break;
            ans++;
        }
    }

    cout << ans;
}

回文は中心を決定すると、それより2長い次の文字列の判定が\(O(1)\)になってよい
今回はそれで\(O(N^2)\)かかって、\(N^2 = 10^8\)で億いったから厳しいかなとか思ってたけど、 定数倍が軽いし、全然余裕で通る(21ms)

using mint = modint<0>;
int main(void){

    int i;

    mint dum; //dum.set_mod(998244353);

    mint ans = 0;
    mint t=0,s=0;
    rep(i,n){
        t += mint(1)/mint(i+1);
    }
    s=t;

    vector<mint> u(n);
    rep(i,n){
        if(i == n-i-1){
            u[i] = s;
        }else if(i > n-i-1) break;
        else{
            u[i] = s;
            u[n-i-1] = s;

            t -= mint(1)/mint(i+1);
            t -= mint(1)/mint(n-i);
            s += t;
        }
    }

    rep(i,n){
        ans += u[i]*a[i];
    }
    cout << ans << "\n";  

    
}
